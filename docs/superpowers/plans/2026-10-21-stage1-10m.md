## Why the controls look like this

The novelty check (`notes/novelty_check_2026-10-21.md`) found two warnings that shape the design:
1. *Topological Sensitivity in Connectome-Constrained Neural Networks* (arXiv 2604.04033): reported topology advantages largely disappear under fair from-scratch initialisation and degree-preserving nulls. **Response:** every parameter shared by two arms starts bit-identical (Task 5); connectome values use one fan-in rule for every arm; the primary null is degree-preserving; each seed draws a fresh shuffle.
2. *conn2res* (Suárez et al. 2024): real connectomes beat degree-preserving nulls in reservoirs, *except the fruit fly*. **Response:** treat a null as a likely, publishable outcome; the decision rules name it explicitly.

Two design traps found while planning:
- **Uneven start.** With zero biases, an edge gets a gradient on the first step only if input activity reaches its source neuron in time and its effect can reach the output. Without per-step normalisation, at 4 steps, only 61% of real-wiring edges are active at the first step, against 73% shuffled, 97% ER and 93% random-IO (`notes/measurements_2026-10-21.md`, corrected table). That is the kind of initialisation confound arXiv 2604.04033 warns about. **Response:** steps = 4 plus normalisation after every step, which measured 100% in every arm. `edge_usage` and `readout_reach` (Task 2) recompute this for every arm and seed, and `scripts/check_reach.py` (Task 11) stops the plan if any arm falls below 90%.
- **Scale differences.** Different wirings give different activity magnitudes. **Response:** parameter-free RMS normalisation of the whole neuron state after every step, identical in every arm. There is deliberately **no** second normalisation over the readout neurons: it would cancel the last step's normalisation and leave 38% of real-wiring edges idle at the start (design C in the corrected table). The per-step normalisation divides each neuron by the activity of all neurons, so the block is not purely edge-local: there is one shared, wiring-independent scale factor per token per step (similar in role to the fly's APL neuron, which inhibits the whole mushroom body). The readout neurons' share of total activity can still differ between wirings; `w_out` is trainable and absorbs a constant factor.

```python
import numpy as np
import pytest

REAL_REGION = Path(__file__).resolve().parent.parent / "data" / "mb_alpn_783.npz"


def hubby_graph(n=120, m=600, seed=0):
    """A small directed graph with a heavy-tailed in-degree, like a connectome."""
    rng = np.random.default_rng(seed)
    weight = 1.0 / np.arange(1, n + 1) ** 0.8      # neuron 0 is the biggest hub
    p = weight / weight.sum()
    keys = set()
    while len(keys) < m:
        pre, post = int(rng.integers(n)), int(rng.choice(n, p=p))
        if pre != post:
            keys.add((pre, post))
    pre, post = np.array(sorted(keys)).T
    return n, pre, post


@pytest.fixture

    n, pre, post = hubby_graph()
    path = tmp_path / "region.npz"
    np.savez(path, root_id=np.arange(n), pre=pre, post=post, w=np.ones(pre.size, np.float32))
```

`src/clm/graphs.py`:
```python
"""Wiring for the connectome block: the real region and its null models.

An arm is a directed edge list over the same N neurons. Edges run pre -> post;
the block computes post activity from pre activity, so the sparse matrix is
indexed [post, pre], and edges are kept sorted by (post, pre), the order torch
calls coalesced.
"""
import hashlib
from dataclasses import dataclass

import numpy as np

ARMS = ("real", "shuf", "er", "real_randio")
K_IO = 256
_ARM_TAG = {"real": 0, "shuf": 1, "er": 2, "real_randio": 3}


@dataclass(frozen=True, eq=False)
class Wiring:
    arm: str
    seed: int
    n: int
    pre: np.ndarray
    post: np.ndarray
    inp: np.ndarray
    out: np.ndarray

    @property
    def nnz(self):
        return int(self.pre.size)

    def fingerprint(self):
        h = hashlib.sha256(f"{self.arm}|{self.n}|".encode())
        for a in (self.pre, self.post, self.inp, self.out):
            h.update(np.ascontiguousarray(a, dtype=np.int64).tobytes())
        return h.hexdigest()


def bad_edges(n, pre, post):
    """True for every self-loop and every repeat of an earlier (pre, post) pair."""
    key = pre.astype(np.int64) * n + post
    order = np.argsort(key, kind="stable")
    s = key[order]
    dup = np.zeros(key.size, bool)
    dup[order[1:]] = s[1:] == s[:-1]
    return dup | (pre == post)


def load_region(path):
    z = np.load(path)
    n = int(len(z["root_id"]))
    pre, post = z["pre"].astype(np.int64), z["post"].astype(np.int64)
    if pre.min() < 0 or max(pre.max(), post.max()) >= n:
        raise ValueError(f"{path}: edge index outside 0..{n - 1}")
    if bad_edges(n, pre, post).any():
        raise ValueError(f"{path}: region has self-loops or duplicate edges")
    return n, pre, post


def degree_preserving_shuffle(n, pre, post, rng, max_passes=20, tries=1000):

    # failed test_region_real.py. Replaced with targeted Maslov-Sneppen swaps;
    # see src/clm/graphs.py for the code as built. 0.9 s, 7.4% of real edges retained.

def erdos_renyi(n, m, rng):
    """m distinct directed edges drawn uniformly, no self-loops."""
    if m > n * (n - 1):
        raise ValueError(f"{m} edges do not fit in {n} neurons")
    keys = np.empty(0, np.int64)
    while keys.size < m:
        k = rng.integers(0, n * n, size=2 * (m - keys.size), dtype=np.int64)
        keys = np.union1d(keys, k[k // n != k % n])
    keys = rng.choice(keys, size=m, replace=False)
    return keys // n, keys % n


def select_io(n, post, k, rng=None):
    """Input = the k lowest in-degree neurons, readout = the k highest.

    This copies the fly-criticality E5/E6 setup where real wiring beat its
    shuffles. With rng, both sets are drawn at random (the real_randio arm).
    """
    if 2 * k > n:
        raise ValueError(f"need {2 * k} distinct neurons, region has {n}")
    if rng is not None:
        pick = rng.choice(n, size=2 * k, replace=False)
        return np.sort(pick[:k]), np.sort(pick[k:])
    order = np.argsort(np.bincount(post, minlength=n), kind="stable")
    return np.sort(order[:k]), np.sort(order[::-1][:k])


def make_wiring(region_path, arm, seed, k=K_IO):
    if arm not in ARMS:
        raise ValueError(f"unknown arm {arm!r}; expected one of {ARMS}")
    n, pre, post = load_region(region_path)
    rng = np.random.default_rng([seed, _ARM_TAG[arm]])
    if arm == "real":
        seed = 0            # the real wiring is the same at every seed
    elif arm == "shuf":
        post = degree_preserving_shuffle(n, pre, post, rng)
    elif arm == "er":
        pre, post = erdos_renyi(n, pre.size, rng)
    inp, out = select_io(n, post, k, rng if arm == "real_randio" else None)
    order = np.lexsort((pre, post))
    return Wiring(arm, seed, n, pre[order], post[order], inp, out)


def forward_reach(w, hops):
    """Neurons reachable from an input neuron in at most `hops` edges."""
    seen = np.zeros(w.n, bool)
    seen[w.inp] = True
    for _ in range(hops):
        seen[w.post[seen[w.pre]]] = True
    return seen


def edge_usage(w, steps):
    """Fraction of edges that get a gradient on the first step, biases at zero.

    Only neurons that input activity has reached are non-zero, and the last
    of the steps-1 sparse passes reads the state after steps-2 hops. Because
    the block normalises over all neurons every step, every lit neuron affects
    the output, so the source neuron being lit is the whole condition.
    """
    return float(forward_reach(w, steps - 2)[w.pre].mean())


def readout_reach(w, steps):
    """Fraction of readout neurons that input activity reaches in steps-1 hops."""
    return float(forward_reach(w, steps - 1)[w.out].mean())
```

Block equations (the spec the reference test encodes), per token x, where N(v) = v / sqrt(mean over all neurons of v² + 1e-6):
- drive = bias + scatter of `w_in·x` onto the input neurons
- h₁ = N(leak · gelu(drive))
- h_{s+1} = N((1 − leak) · h_s + leak · gelu(A h_s + drive)), for s = 1 … steps−1
- y = `w_out` · h_steps[readout]

N has no parameters and maps an all-zero state to zero, so a neuron that input has not reached stays exactly zero. There is no extra normalisation on the readout (see "Why the controls look like this").

```python
def reference(b, w, x):
    """The block's equations written out with a dense matrix, one token at a time."""
    a = np.zeros((w.n, w.n))
    a[w.post, w.pre] = b.values.detach().double().numpy()    # row = receiving neuron
    bias = b.bias.detach().double().numpy()
    w_in = b.w_in.weight.detach().double().numpy()
    w_out = b.w_out.weight.detach().double().numpy()
    gelu = lambda z: 0.5 * z * (1 + erf(z / np.sqrt(2)))
    norm = lambda v: v / np.sqrt(np.mean(v ** 2) + 1e-6)     # over all neurons
    out = []
    for tok in x.double().numpy():
        drive = bias.copy()
        drive[w.inp] += w_in @ tok
        h = norm(b.leak * gelu(drive))
        for _ in range(b.steps - 1):
            h = norm((1 - b.leak) * h + b.leak * gelu(a @ h + drive))
        out.append(w_out @ h[w.out])
    return torch.from_numpy(np.array(out))
```

```python
"""Blocks that fill the FFN slot of a transformer layer.

Both run in float32 even under autocast: torch's sparse matmul has no bfloat16
kernel, and running the dense control at a different precision would add a
difference between arms that is not wiring.
"""
import numpy as np
import torch
import torch.nn as nn
import torch.nn.functional as F


class DenseFFN(nn.Module):
    def __init__(self, d_model, hidden):
        super().__init__()
        self.fc = nn.Linear(d_model, hidden)
        self.proj = nn.Linear(hidden, d_model)

    def forward(self, x):
        with torch.autocast(x.device.type, enabled=False):
            return self.proj(F.gelu(self.fc(x.float())))


class ConnectomeFFN(nn.Module):
    """A leaky recurrent layer whose only connections are a wiring's edges.

    Each token drives the input neurons, activity spreads over the wiring for
    steps-1 matrix applications with a fixed leak, and the readout neurons are
    projected back to d_model. Tokens never mix: this fills a position-wise
    FFN slot, so all sequence mixing stays in attention. The state is
    RMS-normalised over all neurons after every step, with no parameters, so
    activity scale cannot differ between wirings. The readout is deliberately
    not normalised again: that would cancel the last step's norm and leave
    many edges without a gradient at the start.
    """

    def __init__(self, d_model, wiring, steps, leak):
        super().__init__()
        if steps < 2:
            raise ValueError("steps must be >= 2, or the wiring is never used")
        self.n, self.steps, self.leak = wiring.n, steps, leak
        self.fingerprint = wiring.fingerprint()
        self.register_buffer("idx", torch.from_numpy(np.stack([wiring.post, wiring.pre])).long())
        self.register_buffer("inp", torch.from_numpy(wiring.inp).long())
        self.register_buffer("out", torch.from_numpy(wiring.out).long())
        indeg = np.bincount(wiring.post, minlength=wiring.n)
        # fan-in init, one rule for every arm: each neuron's summed input
        # starts at unit variance whatever its in-degree
        self.register_buffer("init_std", torch.from_numpy(1.0 / np.sqrt(indeg[wiring.post])).float())
        self.values = nn.Parameter(torch.zeros(wiring.nnz))
        self.bias = nn.Parameter(torch.zeros(wiring.n))
        self.w_in = nn.Linear(d_model, wiring.inp.size, bias=False)
        self.w_out = nn.Linear(wiring.out.size, d_model, bias=False)

    def state(self, x):
        """Neuron activity after the last step, shape (n, tokens), for x of shape (tokens, d)."""
        drive = x.new_zeros(self.n, x.shape[0]).index_add(0, self.inp, self.w_in(x).t())
        drive = drive + self.bias[:, None]
        a = torch.sparse_coo_tensor(self.idx, self.values, (self.n, self.n), is_coalesced=True)
        h = _norm(self.leak * F.gelu(drive))
        for _ in range(self.steps - 1):
            h = _norm((1 - self.leak) * h + self.leak * F.gelu(torch.sparse.mm(a, h) + drive))
        return h

    def forward(self, x):
        shape = x.shape
        with torch.autocast(x.device.type, enabled=False):
            h = self.state(x.reshape(-1, shape[-1]).float())
            y = self.w_out(h.index_select(0, self.out).t())
        return y.reshape(*shape[:-1], y.shape[-1])


def _norm(h):
    """RMS over the neuron axis (dim 0 of an (n, tokens) state), per token."""
    return h * torch.rsqrt(h.pow(2).mean(dim=0, keepdim=True) + 1e-6)


def connectome_param_count(d_model, wiring):
    return wiring.nnz + wiring.n + d_model * (wiring.inp.size + wiring.out.size)


def matched_hidden(d_model, n_params):
    """Hidden width at which a DenseFFN has n_params parameters, rounded."""
    return round((n_params - d_model) / (2 * d_model + 1))
```

```python
"""A small pre-norm GPT whose FFN slots can hold a connectome block."""
import hashlib
import math
from dataclasses import dataclass

import torch
import torch.nn as nn
import torch.nn.functional as F

from clm.block import ConnectomeFFN, DenseFFN, connectome_param_count, matched_hidden
from clm.graphs import make_wiring

RUN_ARMS = ("real", "shuf", "er", "real_randio", "dense")
_RESIDUAL = (".attn.proj.weight", ".ffn.proj.weight", ".ffn.w_out.weight")
_NORM_WEIGHT = ("ln1.weight", "ln2.weight", "ln_f.weight")


@dataclass(frozen=True)
class ModelConfig:
    vocab: int = 8192
    ctx: int = 256
    d_model: int = 256
    n_layer: int = 6
    n_head: int = 4
    conn_layers: tuple = (0, 1, 2, 3, 4, 5)
    steps: int = 4
    leak: float = 0.5
    k_io: int = 256


class Attention(nn.Module):
    def __init__(self, cfg):
        super().__init__()
        self.n_head = cfg.n_head
        self.qkv = nn.Linear(cfg.d_model, 3 * cfg.d_model)
        self.proj = nn.Linear(cfg.d_model, cfg.d_model)

    def forward(self, x):
        b, t, d = x.shape
        q, k, v = self.qkv(x).view(b, t, 3, self.n_head, d // self.n_head).permute(2, 0, 3, 1, 4)
        y = F.scaled_dot_product_attention(q, k, v, is_causal=True)
        return self.proj(y.transpose(1, 2).reshape(b, t, d))


class Layer(nn.Module):
    def __init__(self, cfg, ffn):
        super().__init__()
        self.ln1, self.ln2 = nn.LayerNorm(cfg.d_model), nn.LayerNorm(cfg.d_model)
        self.attn, self.ffn = Attention(cfg), ffn

    def forward(self, x):
        x = x + self.attn(self.ln1(x))
        return x + self.ffn(self.ln2(x))


class GPT(nn.Module):
    def __init__(self, cfg, ffns):
        super().__init__()
        self.cfg = cfg
        self.tok = nn.Embedding(cfg.vocab, cfg.d_model)
        self.pos = nn.Embedding(cfg.ctx, cfg.d_model)
        self.layers = nn.ModuleList(Layer(cfg, f) for f in ffns)
        self.ln_f = nn.LayerNorm(cfg.d_model)

    def forward(self, idx, targets=None):
        x = self.tok(idx) + self.pos(torch.arange(idx.shape[1], device=idx.device))
        for layer in self.layers:
            x = layer(x)
        logits = self.ln_f(x) @ self.tok.weight.t()          # tied embeddings
        if targets is None:
            return logits, None
        return logits, F.cross_entropy(logits.float().reshape(-1, logits.shape[-1]),
                                       targets.reshape(-1))


def _seed_for(seed, name):
    digest = hashlib.sha256(f"{seed}|{name}".encode()).digest()
    return int.from_bytes(digest[:8], "little") & (2**63 - 1)


def init_weights(model, seed):
    """Every parameter gets its own generator keyed by (seed, name), so a
    parameter that exists in two arms starts identical in both, whatever else
    each arm builds."""
    resid_std = 0.02 / math.sqrt(2 * model.cfg.n_layer)
    with torch.no_grad():
        for name, p in model.named_parameters():
            g = torch.Generator().manual_seed(_seed_for(seed, name))
            if name.endswith(".values"):
                ffn = model.get_submodule(name.rsplit(".", 1)[0])
                new = torch.randn(p.shape, generator=g) * ffn.init_std.cpu()
            elif p.dim() == 1:
                new = torch.ones(p.shape) if name.endswith(_NORM_WEIGHT) else torch.zeros(p.shape)
            else:
                std = resid_std if name.endswith(_RESIDUAL) else 0.02
                new = torch.randn(p.shape, generator=g) * std
            p.copy_(new)


def build_model(cfg, arm, seed, region):
    if arm not in RUN_ARMS:
        raise ValueError(f"unknown arm {arm!r}; expected one of {RUN_ARMS}")
    if arm == "dense":
        ref = make_wiring(region, "real", seed, cfg.k_io)
        hidden = matched_hidden(cfg.d_model, connectome_param_count(cfg.d_model, ref))
        slot = lambda: DenseFFN(cfg.d_model, hidden)
    else:
        wiring = make_wiring(region, arm, seed, cfg.k_io)
        slot = lambda: ConnectomeFFN(cfg.d_model, wiring, cfg.steps, cfg.leak)
    ffns = [slot() if i in cfg.conn_layers else DenseFFN(cfg.d_model, 4 * cfg.d_model)
            for i in range(cfg.n_layer)]
    model = GPT(cfg, ffns)
    init_weights(model, seed)
    return model
```

```python
"""Split rule, tokenizer and encoding for the FineWeb-Edu token files.

scripts/prepare_data.py drives these over a downloaded shard.
"""
import hashlib

import numpy as np

EOT = "<|eot|>"


def split_of(doc_id):
    """1 in 100 documents is validation, chosen by a hash of its id, so the
    split never depends on file order."""
    return "val" if int(hashlib.sha256(doc_id.encode()).hexdigest()[:8], 16) % 100 == 0 else "train"


def train_tokenizer(texts, vocab):
    from tokenizers import Tokenizer, decoders, models, pre_tokenizers, trainers

    tok = Tokenizer(models.BPE())
    tok.pre_tokenizer = pre_tokenizers.ByteLevel(add_prefix_space=False)
    tok.decoder = decoders.ByteLevel()
    trainer = trainers.BpeTrainer(vocab_size=vocab, special_tokens=[EOT],
                                  initial_alphabet=pre_tokenizers.ByteLevel.alphabet())
    tok.train_from_iterator(texts, trainer)
    return tok


def encode_docs(tok, texts):
    """Token ids for each text with the end-of-text id appended, as uint16."""
    if any(EOT in t for t in texts):
        raise ValueError("a document contains the end-of-text marker; filter it first")
    eot = tok.token_to_id(EOT)
    out = []
    for enc in tok.encode_batch(texts):
        ids = np.asarray(enc.ids + [eot], dtype=np.int64)
        if ids.max() >= 2**16:
            raise ValueError(f"token id {ids.max()} does not fit in uint16")
        out.append(ids.astype(np.uint16))
    return out
```

```python
"""Download FineWeb-Edu sample-10BT shard 000 and build the token files.

Writes data/fineweb/{train,val}.bin (uint16), tokenizer.json,
{train,val}_ids.txt and manifest.json. The shard sits in data/raw/, its own
directory: it is downloaded data and is only ever read as parquet.
"""
import hashlib
import json
import sys
import urllib.request
from pathlib import Path

import numpy as np
import pyarrow.parquet as pq

from clm.prep import EOT, encode_docs, split_of, train_tokenizer

ROOT = Path(__file__).resolve().parent.parent
REPO = "HuggingFaceFW/fineweb-edu"
SHARD = "sample/10BT/000_00000.parquet"
URL = f"https://huggingface.co/datasets/{REPO}/resolve/main/{SHARD}"
TRAIN_TOKENS = 200_000_001        # the run reads 199,999,488 + 1 for the last target
VAL_TOKENS = 2_097_153
TOK_DOCS = 50_000
VOCAB = 8192


def lfs_sha256(path):
    api = f"https://huggingface.co/api/datasets/{REPO}/tree/main/{path.rsplit('/', 1)[0]}"
    with urllib.request.urlopen(api, timeout=60) as r:
        for f in json.load(r):
            if f["path"] == path:
                return f["lfs"]["oid"]
    sys.exit(f"{path} not listed by the Hugging Face API")


def sha256(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


def download(url, dest, expected):
    if dest.exists() and sha256(dest) == expected:
        return
    tmp = dest.with_suffix(".part")
    with urllib.request.urlopen(url, timeout=60) as r, open(tmp, "wb") as f:
        while chunk := r.read(1 << 22):
            f.write(chunk)
    got = sha256(tmp)
    if got != expected:
        tmp.unlink()
        sys.exit(f"download hash {got} != expected {expected}")
    tmp.replace(dest)


def docs(path):
    pf = pq.ParquetFile(path)
    missing = {"id", "text"} - set(pf.schema_arrow.names)
    if missing:
        sys.exit(f"shard lacks columns {missing}")
    for b in pf.iter_batches(batch_size=1024, columns=["id", "text"]):
        yield from zip(b.column("id").to_pylist(), b.column("text").to_pylist())


def main():
    raw, out = ROOT / "data" / "raw", ROOT / "data" / "fineweb"
    raw.mkdir(parents=True, exist_ok=True)
    out.mkdir(parents=True, exist_ok=True)
    shard = raw / "000_00000.parquet"
    expected = lfs_sha256(SHARD)
    download(URL, shard, expected)

    tok_texts = []
    for doc_id, text in docs(shard):
        if split_of(doc_id) == "train" and EOT not in text:
            tok_texts.append(text)
            if len(tok_texts) == TOK_DOCS:
                break
    tok = train_tokenizer(iter(tok_texts), VOCAB)
    tok.save(str(out / "tokenizer.json"))

    counts, skipped = {"train": 0, "val": 0}, 0
    targets = {"train": TRAIN_TOKENS, "val": VAL_TOKENS}
    files = {s: open(out / f"{s}.bin", "wb") for s in counts}
    ids = {s: open(out / f"{s}_ids.txt", "w") for s in counts}
    pending = []

    def flush():
        enc = encode_docs(tok, [t for _, _, t in pending])
        for (split, doc_id, _), arr in zip(pending, enc):
            if counts[split] < targets[split]:
                arr.tofile(files[split])
                ids[split].write(doc_id + "\n")
                counts[split] += arr.size
        pending.clear()

    for doc_id, text in docs(shard):
        if all(counts[s] >= targets[s] for s in counts):
            break
        if EOT in text:
            skipped += 1
            continue
        pending.append((split_of(doc_id), doc_id, text))
        if len(pending) == 1024:
            flush()
    flush()
    for f in (*files.values(), *ids.values()):
        f.close()
    short = {s: (counts[s], targets[s]) for s in counts if counts[s] < targets[s]}
    if short:
        sys.exit(f"shard ran out before the targets were met: {short}")

    manifest = {"shard": URL, "shard_sha256": expected, "vocab": VOCAB, "tokenizer_docs": TOK_DOCS,
                "tokenizer_sha256": sha256(out / "tokenizer.json"),
                "split_rule": "val iff int(sha256(id)[:8], 16) % 100 == 0",
                "tokens": counts, "skipped_docs_with_eot": skipped,
                "licence": "ODC-By 1.0 (FineWeb-Edu)"}
    (out / "manifest.json").write_text(json.dumps(manifest, indent=2) + "\n")
    print(json.dumps(manifest, indent=2))


if __name__ == "__main__":
    main()
```

`src/clm/data.py`:
```python
"""Token files and batch sampling.

Batch contents depend only on (seed, step), never on the arm or on where a
run resumed, so every arm at a given seed sees the same tokens in the same
order. The train file is cut into non-overlapping windows and read once, in
a seeded random order: no window is seen twice.
"""
from functools import lru_cache

import numpy as np
import torch


def open_tokens(path):
    return np.memmap(path, dtype=np.uint16, mode="r")


@lru_cache(maxsize=4)
def _order(windows, seed):
    return np.random.default_rng(seed).permutation(windows)


def batch_starts(n_tokens, ctx, batch, seed, step):
    order = _order((n_tokens - 1) // ctx, seed)
    sel = order[step * batch:(step + 1) * batch]
    if sel.size < batch:
        raise ValueError(f"step {step} runs past one epoch of the train file")
    return sel.astype(np.int64) * ctx


def get_batch(tokens, starts, ctx, device):
    x = np.stack([tokens[s:s + ctx] for s in starts]).astype(np.int64)
    y = np.stack([tokens[s + 1:s + ctx + 1] for s in starts]).astype(np.int64)
    return torch.from_numpy(x).to(device), torch.from_numpy(y).to(device)


def eval_windows(n_tokens, ctx, n_eval):
    """Start offsets of the fixed, non-overlapping validation windows."""
    return np.arange(min(n_eval, n_tokens - 1) // ctx, dtype=np.int64) * ctx
```

```python
    if step < cfg.warmup:
        return cfg.lr * (step + 1) / cfg.warmup
    frac = min(1.0, (step - cfg.warmup) / max(1, steps - 1 - cfg.warmup))
    return cfg.min_lr + 0.5 * (cfg.lr - cfg.min_lr) * (1 + math.cos(math.pi * frac))


def load_config(path):
    raw = json.loads(Path(path).read_text())
    model = {**raw["model"], "conn_layers": tuple(raw["model"]["conn_layers"])}
    return ModelConfig(**model), TrainConfig(**raw["train"])


def _sha256(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest() if path and Path(path).exists() else None


def _git():
    def git(*args):
        return subprocess.run(["git", "-C", str(ROOT), *args], capture_output=True, text=True)
    head = git("rev-parse", "HEAD")
    if head.returncode != 0:
        return None, None
    return head.stdout.strip(), bool(git("status", "--porcelain").stdout.strip())


def _autocast(device):
    return torch.autocast(device.type, dtype=torch.bfloat16, enabled=device.type == "cuda")


def param_groups(model, wd):
    decay, rest = [], []
    for name, p in model.named_parameters():
        (decay if p.dim() >= 2 or name.endswith(".values") else rest).append(p)
    return [{"params": decay, "weight_decay": wd}, {"params": rest, "weight_decay": 0.0}]


@torch.no_grad()
def evaluate(model, val, ctx, n_eval, micro, device):
    model.eval()
    starts = eval_windows(len(val), ctx, n_eval)
    total = 0.0
    for i in range(0, len(starts), micro):
        x, y = get_batch(val, starts[i:i + micro], ctx, device)
        with _autocast(device):
            _, loss = model(x, y)
        total += loss.item() * len(x)
    model.train()
    return total / len(starts)


def _write(path, obj):
    tmp = path.with_suffix(".tmp")
    tmp.write_text(json.dumps(obj, indent=2) + "\n")
    tmp.replace(path)


def run(arm, seed, mcfg, tcfg, out, device, max_steps=None, config_path=None):
    out = Path(out)
    out.mkdir(parents=True, exist_ok=True)
    region = ROOT / tcfg.region
    model = build_model(mcfg, arm, seed, region).to(device)
    opt = torch.optim.AdamW(param_groups(model, tcfg.wd), lr=tcfg.lr, betas=(0.9, 0.95))
    steps = tcfg.tokens // (tcfg.batch * mcfg.ctx)
    ckpt, start = out / "ckpt.pt", 0
    if ckpt.exists():
        state = torch.load(ckpt, map_location=device)
        if (state["arm"], state["seed"]) != (arm, seed):
            raise RuntimeError(f"checkpoint in {out} is {state['arm']}-{state['seed']}, not {arm}-{seed}")
        model.load_state_dict(state["model"])
        opt.load_state_dict(state["opt"])
        start = state["step"]
    train, val = open_tokens(ROOT / tcfg.train_bin), open_tokens(ROOT / tcfg.val_bin)
    accum, end = tcfg.batch // tcfg.micro, steps if max_steps is None else min(steps, max_steps)
    val_loss = None
    with open(out / "log.jsonl", "a") as log:
        for step in range(start, end):
            t0 = time.time()
            lr = lr_at(step, steps, tcfg)
            for g in opt.param_groups:
                g["lr"] = lr
            starts = batch_starts(len(train), mcfg.ctx, tcfg.batch, seed, step)
            loss_sum = 0.0
            for k in range(accum):
                x, y = get_batch(train, starts[k * tcfg.micro:(k + 1) * tcfg.micro], mcfg.ctx, device)
                with _autocast(device):
                    _, loss = model(x, y)
                (loss / accum).backward()
                loss_sum += loss.item() / accum
            gnorm = torch.nn.utils.clip_grad_norm_(model.parameters(), tcfg.clip).item()
            if not (math.isfinite(loss_sum) and math.isfinite(gnorm)):
                _write(out / "status.json", {"status": "diverged", "step": step,
                                             "loss": loss_sum, "grad_norm": gnorm})
                raise SystemExit(2)
            opt.step()
            opt.zero_grad(set_to_none=True)
            rec = {"step": step, "loss": loss_sum, "lr": lr, "grad_norm": gnorm,
                   "seconds": round(time.time() - t0, 3)}
            if (step + 1) % tcfg.eval_every == 0 or step + 1 == steps:
                val_loss = rec["val_loss"] = evaluate(model, val, mcfg.ctx, tcfg.eval_tokens,
                                                       tcfg.micro, device)
            log.write(json.dumps(rec) + "\n")
            log.flush()
            if (step + 1) % tcfg.ckpt_every == 0 or step + 1 == end:
                torch.save({"arm": arm, "seed": seed, "step": step + 1,
                            "model": model.state_dict(), "opt": opt.state_dict()}, out / "ckpt.tmp")
                (out / "ckpt.tmp").replace(ckpt)
    if end < steps:
        return
    commit, dirty = _git()
    _write(out / "results.json", {
        "status": "complete", "arm": arm, "seed": seed,
        "fingerprints": sorted({m.fingerprint for m in model.modules() if isinstance(m, ConnectomeFFN)}),
        "params_total": sum(p.numel() for p in model.parameters()),
        "tokens_seen": steps * tcfg.batch * mcfg.ctx, "steps": steps,
        "final_val_loss": val_loss,
        "config_sha256": _sha256(config_path), "prereg_sha256": _sha256(ROOT / "PREREG.md"),
        "git_commit": commit, "git_dirty": dirty,
        "device": torch.cuda.get_device_name(device) if device.type == "cuda" else "cpu"})
```

```python
"""Turn finished runs into the pre-registered verdicts (PREREG.md section 5).

A run that is missing, incomplete, non-finite, short of the token budget, or
whose wiring fingerprint does not match the wiring its arm and seed should
have, is listed as invalid and never dropped silently: any invalid run makes
every contrast that uses its arm INVALID.
"""
import argparse
import json
import math
from pathlib import Path

from clm.graphs import make_wiring

SESOI = 0.01
CONTRASTS = (("primary", "real", "shuf"),          # does the real wiring matter?
             ("degree_stats", "shuf", "er"),       # do degree statistics matter?
             ("io_placement", "real", "real_randio"),
             ("vs_dense", "real", "dense"))


def check_run(result, tokens, fingerprint):
    if result is None:
        return ["missing results.json"]
    problems = []
    if result.get("status") != "complete":
        problems.append(f"status is {result.get('status')!r}")
    loss = result.get("final_val_loss")
    if not isinstance(loss, (int, float)) or not math.isfinite(loss):
        problems.append(f"non-finite final_val_loss {loss!r}")
    if result.get("tokens_seen") != tokens:
        problems.append(f"tokens_seen {result.get('tokens_seen')} != {tokens}")
    if fingerprint is None and result.get("fingerprints"):
        problems.append("dense run carries a wiring")
    elif fingerprint is not None and result.get("fingerprints") != [fingerprint]:
        problems.append("wiring fingerprint does not match this arm and seed")
    return problems


def verdict(x, y, sesoi=SESOI):
    """Paired by seed: is loss x lower than loss y? (dicts seed -> loss)"""
    seeds = sorted(x)
    if sorted(y) != seeds or len(seeds) < 3:
        return "INVALID"
    d = [x[s] - y[s] for s in seeds]
    mean = sum(d) / len(d)
    if mean <= -sesoi and all(v < 0 for v in d):
        return "X_BETTER"
    if mean >= sesoi and all(v > 0 for v in d):
        return "X_WORSE"
    if abs(mean) < sesoi and max(abs(v) for v in d) < 2 * sesoi:
        return "NO_MEANINGFUL_DIFFERENCE"
    return "INCONCLUSIVE"


def analyze(runs_dir, region, tokens, seeds, k_io=256):
    runs_dir = Path(runs_dir)
    arms = sorted({a for _, x, y in CONTRASTS for a in (x, y)})
    loss, invalid = {a: {} for a in arms}, {}
    for arm in arms:
        for seed in seeds:
            name = f"{arm}-{seed}"
            path = runs_dir / name / "results.json"
            result = json.loads(path.read_text()) if path.exists() else None
            fp = None if arm == "dense" else make_wiring(region, arm, seed, k_io).fingerprint()
            problems = check_run(result, tokens, fp)
            if problems:
                invalid[name] = problems
            else:
                loss[arm][seed] = result["final_val_loss"]
    contrasts = {}
    for name, x, y in CONTRASTS:
        bad = any(n.rsplit("-", 1)[0] in (x, y) for n in invalid)
        diffs = {s: loss[x][s] - loss[y][s] for s in seeds if s in loss[x] and s in loss[y]}
        contrasts[name] = {"x": x, "y": y, "diffs": diffs,
                           "verdict": "INVALID" if bad else verdict(loss[x], loss[y])}
    return {"sesoi": SESOI, "losses": loss, "invalid_runs": invalid, "contrasts": contrasts}


def main():
    from clm.train import ROOT, load_config
    ap = argparse.ArgumentParser()
    ap.add_argument("--runs", required=True)
    ap.add_argument("--config", required=True)
    args = ap.parse_args()
    mcfg, tcfg = load_config(args.config)
    tokens = (tcfg.tokens // (tcfg.batch * mcfg.ctx)) * tcfg.batch * mcfg.ctx
    report = analyze(args.runs, ROOT / tcfg.region, tokens, (0, 1, 2), mcfg.k_io)
    (Path(args.runs) / "analysis.json").write_text(json.dumps(report, indent=2) + "\n")
    print(json.dumps(report["contrasts"], indent=2))
    if report["invalid_runs"]:
        print("INVALID RUNS:", json.dumps(report["invalid_runs"], indent=2))


if __name__ == "__main__":
    main()
```

```python
"""Time one training step per layout and apply the layout rule (PREREG.md section 3).

Random token ids only: no text, no validation file, no loss is kept. For each
layout and arm it finds the largest micro-batch that fits, times full steps
(64 sequences, with accumulation), and projects GPU-hours for the whole matrix.
Rule: use all six layers if the projected total fits --budget-hours, otherwise
layers (2, 3); if neither fits, exit 3.
"""
import argparse
import json
import sys
import time
from pathlib import Path

import torch

from clm.model import ModelConfig, build_model
from clm.train import TrainConfig, param_groups

ROOT = Path(__file__).resolve().parent.parent
LAYOUTS = ((0, 1, 2, 3, 4, 5), (2, 3))
CONN_RUNS, DENSE_RUNS = 12, 3


def time_step(mcfg, arm, micro, timed, device):
    model = build_model(mcfg, arm, 0, ROOT / "data" / "mb_alpn_783.npz").to(device)
    opt = torch.optim.AdamW(param_groups(model, 0.1), lr=1e-4)
    tc = TrainConfig()
    accum = tc.batch // micro
    x = torch.randint(0, mcfg.vocab, (micro, mcfg.ctx + 1), device=device)

    def step():
        for _ in range(accum):
            with torch.autocast(device.type, dtype=torch.bfloat16):
                _, loss = model(x[:, :-1], x[:, 1:])
            (loss / accum).backward()
        opt.step()
        opt.zero_grad(set_to_none=True)

    for _ in range(2):
        step()
    torch.cuda.synchronize()
    t = time.time()
    for _ in range(timed):
        step()
    torch.cuda.synchronize()
    return (time.time() - t) / timed, torch.cuda.max_memory_allocated() / 1e9


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--budget-hours", type=float, required=True)
    ap.add_argument("--timed", type=int, default=5)
    args = ap.parse_args()
    device = torch.device("cuda")
    steps_per_run = TrainConfig().tokens // (TrainConfig().batch * 256)
    report = {"gpu": torch.cuda.get_device_name(), "steps": ModelConfig().steps, "layouts": {}}
    for layout in LAYOUTS:
        mcfg = ModelConfig(conn_layers=layout)
        row = {}
        for arm in ("real", "dense"):
            for micro in (16, 8, 4, 2):
                torch.cuda.empty_cache()
                torch.cuda.reset_peak_memory_stats()
                try:
                    sec, gb = time_step(mcfg, arm, micro, args.timed, device)
                except torch.OutOfMemoryError:
                    continue
                row[arm] = {"micro": micro, "sec_per_step": sec, "peak_gb": gb,
                            "hours_per_run": sec * steps_per_run / 3600}
                break
            else:
                row[arm] = {"oom": True}
        if all("hours_per_run" in row[a] for a in row):
            row["total_hours"] = (CONN_RUNS * row["real"]["hours_per_run"]
                                  + DENSE_RUNS * row["dense"]["hours_per_run"])
        report["layouts"][str(layout)] = row
    fits = [l for l in LAYOUTS if report["layouts"][str(l)].get("total_hours", 1e9) <= args.budget_hours]
    report["budget_hours"] = args.budget_hours
    report["chosen_layout"] = list(fits[0]) if fits else None
    (ROOT / "results").mkdir(exist_ok=True)
    (ROOT / "results" / "speed.json").write_text(json.dumps(report, indent=2) + "\n")
    print(json.dumps(report, indent=2))
    if not fits:
        sys.exit(3)


if __name__ == "__main__":
    main()
```

```python
"""Run all 15 (arm, seed) runs of the frozen config, skipping completed ones.

Order is seed-major so an interruption leaves whole seeds complete. Prints
only completion status, never a loss: losses are read once, by clm.analyze,
after every run is done.
"""
import json
import subprocess
import sys
from pathlib import Path

from clm.model import RUN_ARMS

ROOT = Path(__file__).resolve().parent.parent
PY = ROOT / ".venv" / "bin" / "python"
CONFIG = ROOT / "configs" / "stage1.json"


def main():
    if subprocess.run(["git", "-C", str(ROOT), "status", "--porcelain"],
                      capture_output=True, text=True).stdout.strip():
        sys.exit("working tree is dirty; commit before production runs")
    failed = []
    for seed in (0, 1, 2):
        for arm in RUN_ARMS:
            out = ROOT / "runs" / f"{arm}-{seed}"
            res = out / "results.json"
            if res.exists() and json.loads(res.read_text()).get("status") == "complete":
                print(f"{arm}-{seed}: already complete")
                continue
            code = subprocess.run([str(PY), "-m", "clm.train", "--arm", arm, "--seed", str(seed),
                                   "--config", str(CONFIG), "--out", str(out)]).returncode
            print(f"{arm}-{seed}: exit {code}")
            if code != 0:
                failed.append(f"{arm}-{seed}")
    if failed:
        sys.exit(f"failed runs: {failed}")


if __name__ == "__main__":
    main()
```
